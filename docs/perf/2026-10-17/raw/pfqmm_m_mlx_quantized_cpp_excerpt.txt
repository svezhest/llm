 1160      int M,
 1161      int N,
 1162      int K,
 1163      metal::Device& d,
 1164      const Stream& s,
 1165      const std::string& mode) {
 1166    // Choose split_k to target ~512 threadgroups
 1167    int bm = 32, bn = 32;
 1168    int n_tiles = (N + bn - 1) / bn;
 1169    int m_tiles = (M + bm - 1) / bm;
 1170    int current_tgs = n_tiles * m_tiles;
 1171    int split_k = std::max(1, 512 / current_tgs);
 1172  
 1173    // Each K partition must be a whole number of BK-wide (32) K-tiles as well as
 1174    // whole quantization groups. The qmm_t_splitk kernels tile K by BK=32 and do
 1175    // not bound the K dimension, so a partition smaller than BK (e.g. nvfp4's
 1176    // group_size=16) would over-read into the next group's weights/scales.
 1177    int k_align = group_size > 32 ? group_size : 32;
 1178    split_k = std::min(split_k, K / k_align);
 1179  
 1180    // Ensure K divides evenly by split_k * k_align
 1181    while (split_k > 1 && (K % (split_k * k_align) != 0)) {
 1182      split_k--;
 1183    }

 1185      return qmm(
 1186          x,
 1187          w,
 1188          scales,
 1189          biases,
 1190          std::nullopt,
 1191          out,
 1192          true,
 1193          group_size,
